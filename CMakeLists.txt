cmake_minimum_required(VERSION 3.16)
project(starter_pistol_desktop LANGUAGES CXX)
include(CTest)

function(add_firmware_test target)
  add_executable(${target} tests/firmware_test.cpp)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE tests/fakes include)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME ${target} COMMAND ${target})
endfunction()

add_firmware_test(firmware_tests)
add_firmware_test(firmware_long_fir_tests)
target_compile_definitions(firmware_long_fir_tests PRIVATE USE_LONG_FIR)
add_firmware_test(arduino_firmware_tests)
target_compile_definitions(arduino_firmware_tests PRIVATE TEST_ARDUINO_SKETCH)
